    public void LoadExperience(string modeName)
    {
        _currentMode = modeName;
        _modeTickCount = 0;
        _uiActiveGameLabel.Text = $"  [Mode: {modeName}]";

        foreach (Node child in _activeGameContainer.GetChildren())
        {
            child.QueueFree();
        }
        WorldBlocks.Clear();

        if (CommunityGames.ContainsKey(modeName))
        {
            foreach (var b in CommunityGames[modeName])
            {
                SpawnBlock(new Vector3(b.X, b.Y, b.Z), new Vector3(2, 0.8f, 2), Color.FromHtml(b.HexColor), b.IsGoal, 25);
            }
            ResetPlayerPosition();
            return;
        }

        switch (modeName)
        {
            case "Sandbox":
            case "Voxel Sandbox":
                BuildVoxelTerrain(16, 4, 16, new Color(0.28f, 0.72f, 0.35f));
                break;
            case "Creator Studio":
                CustomBuildBlocks.Clear();
                SpawnBlock(new Vector3(8, 2, 8), new Vector3(6, 1, 6), new Color(0.25f, 0.28f, 0.35f));
                break;
            case "Mega Sky Obby":
                BuildObbyCourse();
                break;
            case "Coin Rush Simulator":
                BuildVoxelTerrain(20, 1, 20, new Color(0.22f, 0.25f, 0.32f));
                SpawnCoinPickups(20);
                break;
            case "Natural Disaster Survival":
                BuildVoxelTerrain(14, 2, 14, new Color(0.85f, 0.35f, 0.25f));
                break;
            case "Miner's Coin Tycoon":
                BuildTycoonFactory();
                break;
            case "Sword Fight Heights":
                BuildSwordFightIslands();
                break;
            case "Color Block Party":
                BuildColorPartyGrid();
                break;
            case "Speed Run 4":
                BuildSpeedRunTrack();
                break;
            case "Zombie Survival Wave":
                BuildVoxelTerrain(18, 1, 18, new Color(0.18f, 0.28f, 0.18f));
                SpawnZombieWave(6);
                break;
            case "Laser Tag Arena":
                BuildLaserTagArena();
                break;
            case "Roleplay Town":
                BuildRoleplayTown();
                break;
            case "Fishing Simulator":
                BuildFishingLake();
                break;
        }

        ResetPlayerPosition();
    }

    private void ResetPlayerPosition()
    {
        var player = GetNodeOrNull<CharacterBody3D>("1");
        if (player != null) player.Position = new Vector3(8, 8, 8);
    }

    public void PlaceCreatorBlock(Vector3 pos, string hexColor, bool isGoal)
    {
        CustomBlockData data = new CustomBlockData
        {
            X = Mathf.Round(pos.X),
            Y = Mathf.Round(pos.Y),
            Z = Mathf.Round(pos.Z),
            HexColor = hexColor,
            IsGoal = isGoal
        };
        CustomBuildBlocks.Add(data);
        SpawnBlock(new Vector3(data.X, data.Y, data.Z), new Vector3(2, 0.8f, 2), Color.FromHtml(hexColor), isGoal, 25);
    }

    private void SpawnBlock(Vector3 pos, Vector3 size, Color color, bool isRewardPad = false, int rewardAmount = 25)
    {
        StaticBody3D body = new StaticBody3D { Position = pos };
        BoxMesh mesh = new BoxMesh { Size = size };
        StandardMaterial3D mat = new StandardMaterial3D { AlbedoColor = color };
        if (isRewardPad)
        {
            mat.EmissionEnabled = true;
            mat.Emission = color;
        }

        MeshInstance3D inst = new MeshInstance3D { Mesh = mesh, MaterialOverride = mat };
        CollisionShape3D col = new CollisionShape3D { Shape = new BoxShape3D { Size = size } };

        body.AddChild(inst);
        body.AddChild(col);
        _activeGameContainer.AddChild(body);
        WorldBlocks[new Vector3I((int)pos.X, (int)pos.Y, (int)pos.Z)] = inst;

        if (isRewardPad)
        {
            Area3D trigger = new Area3D();
            CollisionShape3D trigCol = new CollisionShape3D { Shape = new BoxShape3D { Size = size + new Vector3(0.2f, 1.5f, 0.2f) } };
            trigger.AddChild(trigCol);
            trigger.BodyEntered += (Node3D b) =>
            {
                if (b is CharacterBody3D)
                {
                    SocialManager?.SendEarnCoins(rewardAmount);
                    LoadExperience("Sandbox");
                }
            };
            body.AddChild(trigger);
        }
    }

    private void BuildVoxelTerrain(int width, int height, int depth, Color topColor)
    {
        for (int x = 0; x < width; x++)
        {
            for (int y = 0; y < height; y++)
            {
                for (int z = 0; z < depth; z++)
                {
                    Color c = (y == height - 1) ? topColor : new Color(0.45f, 0.35f, 0.28f);
                    SpawnBlock(new Vector3(x, y, z), Vector3.One, c);
                }
            }
        }
    }
